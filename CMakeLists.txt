cmake_minimum_required(VERSION 3.20)
project(std_t99_receiver VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

set(SPEAR ${CMAKE_CURRENT_SOURCE_DIR}/vendor/spear)
find_path(FFTW3F_INCLUDE_DIR fftw3.h HINTS "${CMAKE_CURRENT_SOURCE_DIR}/build/fftw" REQUIRED)
if(MINGW AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/build/fftw/libfftw3f-3.dll")
  set(FFTW3F_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/build/fftw/libfftw3f-3.dll")
else()
  find_library(FFTW3F_LIBRARY NAMES fftw3f libfftw3f-3 HINTS "${CMAKE_CURRENT_SOURCE_DIR}/build/fftw" REQUIRED)
endif()
add_library(t99_core STATIC
  src/protocol.cpp src/messages.cpp src/receiver.cpp src/session.cpp src/frame_sync.cpp
  ${SPEAR}/apps/std_t98/dsp/rrc.cpp ${SPEAR}/apps/std_t98/dsp/symbol_sync.cpp
  ${SPEAR}/dsp/src/fir.cpp ${SPEAR}/dsp/src/fftw_planner.cpp)
target_include_directories(t99_core PRIVATE ${FFTW3F_INCLUDE_DIR})
target_link_libraries(t99_core PUBLIC ${FFTW3F_LIBRARY})
target_include_directories(t99_core PUBLIC src src/include ${SPEAR}/apps/std_t98 ${SPEAR}/dsp/include)
if(MSVC)
  target_compile_options(t99_core PUBLIC /utf-8 /permissive-)
endif()

add_library(t99_io STATIC src/input.cpp src/output.cpp src/dashboard.cpp)
find_package(Threads REQUIRED)
target_link_libraries(t99_io PUBLIC t99_core Threads::Threads ${CMAKE_DL_LIBS})
add_executable(std-t99 src/main.cpp)
target_link_libraries(std-t99 PRIVATE t99_io)
if(WIN32)
  target_link_libraries(std-t99 PRIVATE shell32)
endif()
if(MINGW)
  target_link_options(std-t99 PRIVATE -static -static-libgcc -static-libstdc++)
endif()
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/build/fftw/libfftw3f-3.dll")
  add_custom_command(TARGET std-t99 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/build/fftw/libfftw3f-3.dll" "$<TARGET_FILE_DIR:std-t99>")
endif()
install(TARGETS std-t99 RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
  add_executable(t99-tests tests/native/test_main.cpp ${SPEAR}/dsp/src/channelizer.cpp)
  target_include_directories(t99-tests PRIVATE ${FFTW3F_INCLUDE_DIR})
  target_link_libraries(t99-tests PRIVATE t99_io)
  add_test(NAME native COMMAND t99-tests)
  add_test(NAME cli_help COMMAND std-t99 --help)
endif()
